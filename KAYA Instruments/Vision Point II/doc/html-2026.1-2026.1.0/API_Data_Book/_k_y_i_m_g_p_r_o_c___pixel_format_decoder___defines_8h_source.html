49 KYVP_DATA_TYPE_RGBA = 0x05, // This is used for RGBA data, where 'A' is the alpha (or overlay) plane, transmitted in the order red, green, blue, alpha. This has no sub-types. This is defined in Table 31 JIIA CXP standard document.

53 KYVP_DATA_TYPE_INVALID = 0xFF // Invalid data type
